    Types¶

    wordbase.types ¶

    Type definitions for WordBase messages.

    This module defines the data structures used for representing conversations in WordBase. It provides a unified message format that works across all LLM providers, with support for both simple text and multimodal (text + image) content.

    The module includes
    • TextContent: Represents a text content block
    • ImageContent: Represents an image content block
    • ToolUseContent: Represents an assistant tool-call request, for replaying it back into conversation history
    • ToolResultContent: Represents a tool's result, sent back to the model
    • Message: Represents a single message in a conversation
    • Tool: A tool definition offered to the model during generate()
    • ToolCall: A single tool invocation requested by the model
    • Utility functions for converting between dicts and Message objects

    Typical usage example:

    from wordbase.types import Message, TextContent, ImageContent

# Simple text message
msg1 = Message(role="user", content="Hello")

# Multimodal message with text and image
msg2 = Message(
    role="user",
    content=[
        TextContent(type="text", text="What's in this image?"),
        ImageContent(type="image", source="path/to/image.jpg")
    ]
)

# Convert from dictionary
msg3 = Message.from_dict({"role": "assistant", "content": "Hi there!"})

    By default, every TextContent/ImageContent/Message validates its fields the moment it is constructed (whether built directly or via from_dict()/ messages_from_dicts()), raising ValueError/TypeError immediately on bad data. Pass validate=False to any constructor to skip this, e.g. when reconstructing data that has already been validated once.

def from_dict(cls, data: dict[str, Any], *, validate: bool = True) -> "Message":
    """Create a Message from a dictionary.

    Args:
        data: Dictionary with 'role' and 'content' keys.
              Content can be a string or a list of content block dicts.
        validate: Whether to validate the resulting Message (and any
            content blocks within it). Defaults to True.

    Returns:
        Message object.

    Raises:
        KeyError: If required keys are missing.
        ValueError: If role or content block types are invalid.
        TypeError: If data structure is invalid.
    """
    role = data["role"]
    content_raw = data["content"]

    # Handle string content
    if isinstance(content_raw, str):
        return cls(role=role, content=content_raw, validate=validate)

    # Handle list of content blocks
    if isinstance(content_raw, list):
        content_blocks: list[ContentBlock] = []
        for block in content_raw:
            if not isinstance(block, dict):
                raise TypeError(f"Content block must be a dict, got {type(block)}")

            block_type = block.get("type")

            if block_type == "text":
                content_blocks.append(
                    TextContent(type="text", text=block["text"], validate=validate)
                )
            elif block_type == "image":
                content_blocks.append(
                    ImageContent(
                        type="image", source=block["source"], validate=validate
                    )
                )
            elif block_type == "tool_use":
                content_blocks.append(
                    ToolUseContent(
                        type="tool_use",
                        id=block["id"],
                        name=block["name"],
                        arguments=block["arguments"],
                        validate=validate,
                    )
                )
            elif block_type == "tool_result":
                content_blocks.append(
                    ToolResultContent(
                        type="tool_result",
                        tool_call_id=block["tool_call_id"],
                        content=block["content"],
                        is_error=block.get("is_error", False),
                        validate=validate,
                    )
                )
            else:
                raise ValueError(f"Invalid content block type: {block_type}")

        return cls(role=role, content=content_blocks, validate=validate)

    raise TypeError(f"Content must be str or list, got {type(content_raw)}")

    def to_dict(self) -> dict[str, Any]:
    """Convert Message to a dictionary.

    Returns:
        Dictionary representation of the message.
    """
    # Handle string content
    if isinstance(self.content, str):
        return {"role": self.role, "content": self.content}

    # Handle list of content blocks
    content_list: list[dict[str, Any]] = []
    for block in self.content:
        if isinstance(block, TextContent):
            content_list.append({"type": "text", "text": block.text})
        elif isinstance(block, ImageContent):
            content_list.append({"type": "image", "source": block.source})
        elif isinstance(block, ToolUseContent):
            content_list.append(
                {
                    "type": "tool_use",
                    "id": block.id,
                    "name": block.name,
                    "arguments": block.arguments,
                }
            )
        elif isinstance(block, ToolResultContent):
            content_list.append(
                {
                    "type": "tool_result",
                    "tool_call_id": block.tool_call_id,
                    "content": block.content,
                    "is_error": block.is_error,
                }
            )

    return {"role": self.role, "content": content_list}

    def messages_from_dicts(
    dicts: list[dict[str, Any]], *, validate: bool = True
) -> list[Message]:
    """Convert a list of dictionaries to Message objects.

    Args:
        dicts: List of message dictionaries.
        validate: Whether to validate each resulting Message. Defaults to
            True. Forwarded to Message.from_dict().

    Returns:
        List of Message objects.

    Raises:
        KeyError: If a dictionary is missing required keys (delegated to
            Message.from_dict()).
    """
    return [Message.from_dict(d, validate=validate) for d in dicts]
